# The host/OpenMP variants still share .cu sources and require CUDA to build.
if(PLATFORM STREQUAL "CUDA")
    foreach(variant IN ITEMS HOST CUDA OPENMP UNIFIED_HOST UNIFIED_CUDA UNIFIED_OPENMP)
        string(TOLOWER "${variant}" suffix)
        string(REPLACE "openmp" "omp" suffix "${suffix}")
        foreach(operation IN ITEMS for_each for_each_nd)
            set(name test_${operation}_${suffix})
            add_executable(${name}.bin test_${operation}.cu)
            target_compile_definitions(${name}.bin PRIVATE TEST_${variant})
            if(variant MATCHES "OPENMP$")
                scfd_target_openmp(${name}.bin)
            endif()
            scfd_add_test(${name} for_each CUDA)
        endforeach()
    endforeach()
endif()

if(PLATFORM STREQUAL "SYCL")
    foreach(operation IN ITEMS for_each for_each_nd)
        set(name test_${operation}_sycl)
        add_executable(${name}.bin ${name}.cpp)
        scfd_target_sycl(${name}.bin)
        scfd_add_test(${name} for_each SYCL)
    endforeach()
endif()

if(PLATFORM STREQUAL "HIP")
    add_executable(test_for_each_hip.bin test_for_each_hip.cpp)
    scfd_add_test(test_for_each_hip for_each HIP)
endif()

add_executable(test_for_each_func_macro.bin test_for_each_func_macro.cpp)
