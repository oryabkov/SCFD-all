cmake_minimum_required(VERSION 3.14)
project(SCFDCapabilityProbe LANGUAGES C CXX)

option(SCFD_WITH_CUDA "Probe CUDA" OFF)
option(SCFD_WITH_HIP "Probe HIP" OFF)
option(SCFD_WITH_SYCL "Probe SYCL" OFF)
option(SCFD_CI_PROBE_MPI "Probe MPI independently of the local backend" OFF)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

if(SCFD_CI_PROBE_MPI)
    find_package(MPI REQUIRED COMPONENTS CXX)
    if(NOT MPIEXEC_EXECUTABLE)
        message(FATAL_ERROR "MPI capability detection requires an MPI launcher")
    endif()
    add_executable(mpi-capability.bin mpi.cpp)
    target_link_libraries(mpi-capability.bin PRIVATE MPI::MPI_CXX)

    # Lists are written one argument per line; the runner reads them with mapfile.
    # Spaces within an individual launcher argument are therefore preserved.
    file(WRITE "${CMAKE_BINARY_DIR}/executable.txt" "${MPIEXEC_EXECUTABLE}\n")
    if(MPI_CXX_COMPILER)
        file(WRITE "${CMAKE_BINARY_DIR}/cxx_compiler.txt" "${MPI_CXX_COMPILER}\n")
    else()
        # FindMPI can discover MPI built into the selected C++ compiler itself.
        file(WRITE "${CMAKE_BINARY_DIR}/cxx_compiler.txt" "${CMAKE_CXX_COMPILER}\n")
    endif()
    file(WRITE "${CMAKE_BINARY_DIR}/numproc_flag.txt" "${MPIEXEC_NUMPROC_FLAG}\n")
    foreach(kind preflags postflags)
        string(TOUPPER "${kind}" upper_kind)
        file(WRITE "${CMAKE_BINARY_DIR}/${kind}.txt" "")
        foreach(flag IN LISTS MPIEXEC_${upper_kind})
            file(APPEND "${CMAKE_BINARY_DIR}/${kind}.txt" "${flag}\n")
        endforeach()
    endforeach()
else()
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SCFDToolchains.cmake")
    scfd_setup_toolchains()
    if(SCFD_WITH_CUDA)
        add_executable(capability.bin cuda.cu)
    elseif(SCFD_WITH_HIP)
        add_executable(capability.bin hip.cpp)
    elseif(SCFD_WITH_SYCL)
        add_executable(capability.bin sycl.cpp)
        scfd_target_sycl(capability.bin)
    else()
        add_executable(capability.bin host.cpp)
        scfd_target_openmp(capability.bin)
    endif()
endif()
